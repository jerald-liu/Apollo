Duration bin contract (apollo/tokenizer/bins.py):

```python
import numpy as np
DURATION_EDGES = np.logspace(np.log10(0.030), np.log10(1.500), num=25)  # 25 edges → 24 bins

def quantize_duration(d_sec: float) -> int:
    d = max(0.030, min(1.500, d_sec))
    return int(np.searchsorted(DURATION_EDGES, d, side='right')) - 1   # → 0..23

def decode_duration(bin_i: int) -> float:
    lo, hi = DURATION_EDGES[bin_i], DURATION_EDGES[bin_i+1]
    return float(np.sqrt(lo * hi))
```

Time and velocity quantization (also in bins.py):

```python
# Time: 32 bins at 120 bpm; 32nd-note width = (60 / 120) / 8 = 0.03125 s; max = 32 * 0.03125 = 1.0 s
def quantize_time_shift(dt_sec: float, tempo_bpm: float = 120.0, n_bins: int = 32) -> int:
    bin_width = (60.0 / tempo_bpm) / 8.0
    bin_i = int(round(dt_sec / bin_width))
    if bin_i < 0 or bin_i >= n_bins:
        raise ValueError(f"time_shift {dt_sec:.3f}s out of bin range [0, {n_bins * bin_width:.3f}s)")
    return bin_i

def decode_time_shift(bin_i: int, tempo_bpm: float = 120.0) -> float:
    bin_width = (60.0 / tempo_bpm) / 8.0
    return bin_i * bin_width

# Velocity: 16 linear bins over 1..127. bin = (vel - 1) * 16 // 127 clamped to 0..15
def quantize_velocity(vel: int) -> int:
    v = max(1, min(127, int(vel)))
    return min(15, (v - 1) * 16 // 127)

def decode_velocity(bin_i: int) -> int:
    # decode to bin center (1-based MIDI velocity)
    return int(round(1 + (bin_i + 0.5) * 127 / 16))
```
