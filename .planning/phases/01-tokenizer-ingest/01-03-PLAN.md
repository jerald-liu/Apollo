    ```python
    """Mel-feature extraction. See .planning/phases/01-tokenizer-ingest/01-RESEARCH.md §Mel Pipeline."""
    import os
    import numpy as np
    import torch
    import torchaudio
    from torchaudio.transforms import Resample, MelSpectrogram

    from apollo.ingest.errors import IngestError


    class MelExtractor:
        TARGET_SR = 22050
        N_FFT = 2048
        HOP_LENGTH = 512
        N_MELS = 128
        TARGET_FRAMES = 96
        LOG_FLOOR = 1e-8
        MAX_WAV_BYTES = 10 * 1024 * 1024      # 10 MB
        MAX_WAV_SECONDS = 30.0                # 30 s

        def __init__(self):
            self.mel = MelSpectrogram(
                sample_rate=self.TARGET_SR,
                n_fft=self.N_FFT,
                hop_length=self.HOP_LENGTH,
                n_mels=self.N_MELS,
                power=2.0,
                center=True,
                mel_scale='htk',
            )
            self._resamplers = {}

        def __call__(self, wav_path: str, pair_path: str) -> torch.Tensor:
            # Security: cap file size before loading
            try:
                size = os.path.getsize(wav_path)
            except OSError as e:
                raise IngestError(pair_path, f"failed to stat {wav_path}: {e}")
            if size > self.MAX_WAV_BYTES:
                raise IngestError(pair_path, f"wav exceeds size cap ({size} > {self.MAX_WAV_BYTES} bytes)")

            try:
                wav, sr = torchaudio.load(wav_path)        # (channels, samples), float32
            except Exception as e:
                raise IngestError(pair_path, f"failed to load {wav_path}: {e}")

            # Security: cap duration
            duration = wav.shape[-1] / float(sr)
            if duration > self.MAX_WAV_SECONDS:
                raise IngestError(pair_path, f"wav duration {duration:.2f}s exceeds cap ({self.MAX_WAV_SECONDS}s)")

            # Mono mix
            if wav.shape[0] > 1:
                wav = wav.mean(dim=0, keepdim=True)

            # Resample (cached per source SR)
            if sr != self.TARGET_SR:
                if sr not in self._resamplers:
                    self._resamplers[sr] = Resample(orig_freq=sr, new_freq=self.TARGET_SR)
                wav = self._resamplers[sr](wav)

            # Mel: (1, n_mels, T)
            mel = self.mel(wav)
            log_mel = torch.log(mel + self.LOG_FLOOR)
            log_mel = log_mel.squeeze(0).transpose(0, 1).contiguous()   # (T, 128)

            return self._fix_frames(log_mel, pair_path)

        def _fix_frames(self, log_mel: torch.Tensor, pair_path: str) -> torch.Tensor:
            T, M = log_mel.shape
            if M != self.N_MELS:
                raise IngestError(pair_path, f"mel produced {M} mels, expected {self.N_MELS}")
            if T >= self.TARGET_FRAMES:
                return log_mel[:self.TARGET_FRAMES, :].contiguous()
            pad_value = float(np.log(self.LOG_FLOOR))   # ≈ -18.42
            pad = torch.full((self.TARGET_FRAMES - T, M), pad_value, dtype=log_mel.dtype)
            return torch.cat([log_mel, pad], dim=0).contiguous().to(torch.float32)
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| disk → torchaudio.load | The WAV file on disk could be attacker-controlled or corrupted (e.g. a user-supplied corpus from a shared source). torchaudio parses arbitrary RIFF/WAV headers. |
| MelExtractor → in-memory tensor | After load, the tensor is fully under our control. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-07 | Denial of Service | huge WAV file (e.g. multi-GB) consumed by torchaudio.load → memory exhaustion | mitigate | `os.path.getsize` check rejects files > 10 MB BEFORE torchaudio.load. 10 MB at 44.1 kHz stereo float32 ≈ 30 sec of audio, more than enough headroom for the 1.5–2.2 s gestures we expect. |
| T-01-08 | Denial of Service | malformed WAV with header claiming very long duration → torchaudio allocates large buffer | mitigate | Post-load duration check (`wav.shape[-1] / sr > 30.0` → IngestError). Combined with size cap above, bounds both file size on disk and decoded sample count. |
| T-01-09 | Information Disclosure | torchaudio error messages leak filesystem paths | accept | We deliberately include the wav path in the IngestError reason so the user can fix the file. Single-user local project — not a multi-tenant concern. |
| T-01-10 | Tampering | fuzzed WAV header causing torchaudio internal error | mitigate | Broad `except Exception` around torchaudio.load → IngestError. Bad files abort with the pair path; pipeline does not crash with a stack trace into the CLI. |

</threat_model>
